class VectorMath:
    @staticmethod
    def cosine_similarity(v1, v2):
        if not v1 or not v2 or len(v1) != len(v2): return 0.0
        dot = sum(a * b for a, b in zip(v1, v2))
        mag1 = math.sqrt(sum(a * a for a in v1))
        mag2 = math.sqrt(sum(b * b for b in v2))
        if mag1 == 0 or mag2 == 0: return 0.0
        return dot / (mag1 * mag2)

class Keyring:
    def __init__(self, keys):
        self.keys = list(set(keys))
        env_key = os.environ.get("JANUS_API_KEY")
        if env_key: self.keys.insert(0, env_key)
        self.index = 0
        self.total = len(self.keys)
    def reload(self, new_keys):
        self.keys = list(set(self.keys) | set(new_keys))
        self.total = len(self.keys)
        self.index = 0
        if new_keys: logger.info(f"{ICONS['moon']} KEYRING: Keys updated. Total: {self.total}")
    def get_next(self):
        if self.total == 0: return None
        k = self.keys[self.index]
        self.index = (self.index + 1) % self.total
        return k
    def bury(self, key):
        if key in self.keys:
            self.keys.remove(key)
            self.total = len(self.keys)
            if self.total > 0: self.index = self.index % self.total
            logger.warning(f"{ICONS['skull']} KEYRING: Key died. Remaining: {self.total}")

class JanusFace:
    def __init__(self, keyring, settings):
        self.keyring = keyring; self.settings = settings
        self.models = [settings.get("model_primary"), settings.get("model_backup")]
    async def _api_call(self, url, payload, key):
        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(f"{url}?key={key}", json=payload, timeout=60) as res:
                    if res.status == 200: return await res.json()
                    elif res.status in [401, 403]: return "AUTH_ERR"
                    elif res.status in [429, 500, 503]: return "BUSY"
                    else: return f"ERR_{res.status}"
        except: return "NET_ERR"
    async def analyze_image(self, image_path):
        try:
            mime, _ = mimetypes.guess_type(image_path); mime = mime or "image/jpeg"
            with open(image_path, "rb") as f: enc = base64.b64encode(f.read()).decode("utf-8")
            payload = {"contents": [{"parts": [{"text": "Describe JSON"}, {"inline_data": {"mime_type": mime, "data": enc}}]}], "generationConfig": {"responseMimeType": "application/json"}}
            url = f"https://generativelanguage.googleapis.com/v1beta/models/{self.settings.get('model_vision')}:generateContent"
            key = self.keyring.get_next()
            if not key: return None
            res = await self._api_call(url, payload, key)
            if isinstance(res, dict) and 'candidates' in res: return json.loads(res['candidates'][0]['content']['parts'][0]['text'])
            return None
        except: return None
    async def invoke(self, prompt, sys_inst=None, temp=None, explicit_model=None):
        targets = explicit_model if isinstance(explicit_model, list) else self.models
        for model in targets:
            attempts = 0
            while attempts < 2:
                key = self.keyring.get_next()
                if not key: logger.error(f"{ICONS['error']} NO KEYS"); return "ERROR: NO_KEYS"
                url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent"
                payload = {"contents": [{"parts": [{"text": prompt}]}], "generationConfig": {"temperature": temp}}
                if sys_inst: payload["system_instruction"] = {"parts": [{"text": sys_inst}]}
                res = await self._api_call(url, payload, key)
                if isinstance(res, dict) and 'candidates' in res: return res['candidates'][0]['content']['parts'][0]['text'].strip()
                elif res == "AUTH_ERR": self.keyring.bury(key)
                elif res == "BUSY": await asyncio.sleep(1)
                attempts += 1
        return f"{ICONS['error']} SILENCE"

            for f_name in files:
                f_path = os.path.join(target, f_name); offset_key = f"OFFSET_{f_name}"; current_offset = 0
                async with aiosqlite.connect(self.memory.db_path) as db:
                    cur = await db.execute("SELECT value FROM vault WHERE key_type=?", (offset_key,))
                    row = await cur.fetchone(); 
                    if row: current_offset = int(row[0])
                lines_read = 0; new_offset = current_offset; finished = False
                with open(f_path, 'r', encoding='utf-8', errors='ignore') as f:
                    f.seek(current_offset)
                    while lines_read < 50:
                        line = f.readline()
                        if not line: finished = True; break
                        new_offset = f.tell()
                        if not line.strip() or '"index":' in line: continue
                        try:
                            d = json.loads(line)
                            text = d.get('text') or d.get('source_text') or d.get('opening_text')
                            title = d.get('title', 'Unknown')
                            if text:
                                await self.memory.remember("ANTIVIK", f"KNOWLEDGE: {title} | {text[:500]}")
                                lines_read += 1
                        except: continue
                if lines_read > 0: logger.info(f"{ICONS['beetle']} ANTIVIK: +{lines_read} ({f_name})")
                if new_offset > current_offset:
                     async with aiosqlite.connect(self.memory.db_path) as db:

                         await db.commit()
                if finished: os.remove(f_path)
        except: pass
