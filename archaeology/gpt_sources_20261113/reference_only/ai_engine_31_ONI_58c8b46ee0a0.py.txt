# ==============================================================================
class VectorMath:
    @staticmethod
    def cosine_similarity(v1, v2):
        """Calculates similarity between two vectors (0.0 to 1.0) without numpy."""
        if not v1 or not v2 or len(v1) != len(v2): return 0.0
        dot_product = sum(a * b for a, b in zip(v1, v2))
        magnitude_v1 = math.sqrt(sum(a * a for a in v1))
        magnitude_v2 = math.sqrt(sum(b * b for b in v2))
        if magnitude_v1 == 0 or magnitude_v2 == 0: return 0.0
        return dot_product / (magnitude_v1 * magnitude_v2)

    def save(self):
        try:
            with open(self.filename, 'w', encoding='utf-8') as f:
                json.dump(self.data, f, indent=4)
        except: pass

    def get(self, key, default=None):
        return self.data.get(key, default if default is not None else self.DEFAULT.get(key))

class Keyring:
    def __init__(self, keys):
        self.keys = list(set(keys))
        env_key = os.environ.get("JANUS_API_KEY")
        if env_key: self.keys.insert(0, env_key)
        self.index = 0
        self.total = len(self.keys)
    
    def reload(self, new_keys):
        self.keys = list(set(self.keys) | set(new_keys))
        self.total = len(self.keys)
        self.index = 0

    def get_next(self):
        if self.total == 0: return None
        k = self.keys[self.index]
        self.index = (self.index + 1) % self.total
        return k

    def bury(self, key):
        if key in self.keys:
            self.keys.remove(key)
            self.total = len(self.keys)
            if self.total > 0: self.index = self.index % self.total
            logger.warning(f"{ICONS['skull']} Key buried. Left: {self.total}")

class FastMemory:
    def __init__(self, capacity=30, ttl=600):
        self.cache = cachetools.TTLCache(maxsize=capacity, ttl=ttl)

    def add(self, key, value): self.cache[key] = value

    def get_recent(self, limit=10):
        items = sorted(self.cache.items(), key=lambda x: x[0], reverse=True)
        return items[:limit]

    def clear(self): self.cache.clear()

class JanusHippocampus:
    def __init__(self, db_path="janus.db"):
        self.db_path = db_path

    async def init_db(self):
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute("CREATE TABLE IF NOT EXISTS thoughts (id INTEGER PRIMARY KEY, timestamp TEXT, source TEXT, content TEXT, tags TEXT)")
            await db.execute("CREATE TABLE IF NOT EXISTS graph_state (id INTEGER PRIMARY KEY, data TEXT)")
            await db.execute("CREATE TABLE IF NOT EXISTS code_dna (id INTEGER PRIMARY KEY, timestamp TEXT, version_tag TEXT, file_name TEXT, source_code TEXT, modules_detected TEXT)")
            await db.execute("CREATE TABLE IF NOT EXISTS vault (id INTEGER PRIMARY KEY, key_type TEXT, value TEXT, UNIQUE(value))")
            await db.commit()

    async def store_key(self, api_key):
        try:
            async with aiosqlite.connect(self.db_path) as db:
                await db.execute("INSERT OR IGNORE INTO vault (key_type, value) VALUES (?, ?)", ("google_api", api_key.strip()))
                await db.commit()
            return True
        except: return False

    async def get_all_keys(self):
        try:
            async with aiosqlite.connect(self.db_path) as db:
                cursor = await db.execute("SELECT value FROM vault WHERE key_type='google_api'")
                return [row[0] for row in await cursor.fetchall()]
        except: return []

    async def remember(self, source, content):
        if not isinstance(content, str): content = json.dumps(content, ensure_ascii=False)
        tags = json.dumps(re.findall(r'\w{4,}', content.lower()), ensure_ascii=False)
        try:
            async with aiosqlite.connect(self.db_path) as db:
                await db.execute("INSERT INTO thoughts (timestamp, source, content, tags) VALUES (?, ?, ?, ?)", 
                                 (datetime.now().isoformat(), source, content, tags))
                await db.commit()
        except Exception as e: logger.error(f"DB Error: {e}")

    async def recall_adaptive(self, limit=15):
        async with aiosqlite.connect(self.db_path) as db:
            cursor = await db.execute("SELECT source, content FROM thoughts ORDER BY id DESC LIMIT ?", (limit,))
            return (await cursor.fetchall())[::-1]

    async def load_graph(self):
        try:
            async with aiosqlite.connect(self.db_path) as db:
                cursor = await db.execute("SELECT data FROM graph_state WHERE id=1")
                row = await cursor.fetchone()
                return json.loads(row[0]) if row else {"nodes": [], "links": []}
        except: return {"nodes": [], "links": []}
    
    async def save_graph(self, nodes, links):
        data = json.dumps({"nodes": nodes, "links": links})
        async with aiosqlite.connect(self.db_path) as db:
            await db.execute("INSERT OR REPLACE INTO graph_state (id, data) VALUES (1, ?)", (data,))
            await db.commit()

class JanusNerveUDP(asyncio.DatagramProtocol):
    def __init__(self, nerve): self.nerve = nerve
    def datagram_received(self, data, addr):
        asyncio.create_task(self.nerve._ingest(data.decode('utf-8', 'ignore').strip(), "UDP", addr[0]))

class JanusNerve:
    def __init__(self, engine):
        self.engine = engine
        self.settings = engine.settings.data.get("nerve_config", {})

    async def start(self):
        try:
            port = self.settings.get('syslog_udp_port', 514)
            loop = asyncio.get_running_loop()
            await loop.create_datagram_endpoint(lambda: JanusNerveUDP(self), local_addr=('0.0.0.0', port))
            logger.info(f"{ICONS['nerve']} UDP Listener on {port}")
        except: pass
        
        # HTTP is started by routes.py calling _handle_http_log directly via route

    async def _ingest(self, content, source_type, metadata=None):
        if not content: return
        try:
            for w in self.settings.get("filters", []): 
                if w in content.lower(): return
            await self.engine.memory.remember(f"ROUTER_{source_type}", content)
        except Exception as e: logger.error(f"Ingest error: {e}")

    async def _handle_http_log(self, request):
        try:
            data = await request.text()
            await self._ingest(data, "HTTP", request.remote)
            return web.Response(text="OK")
        except: return web.Response(status=500)

class JanusFace:
    def __init__(self, keyring, settings):
        self.keyring = keyring
        self.settings = settings
        self.models = [
            settings.get("model_primary"),
            settings.get("model_backup")
        ]
        self.embed_model = settings.get("model_embedding", "text-embedding-004")

    async def _api_call(self, url, payload, key):
        async with aiohttp.ClientSession() as session:
            async with session.post(f"{url}?key={key}", json=payload, timeout=20) as res:
                if res.status == 200:
                    return await res.json()
                elif res.status in [401, 403]:
                    return "AUTH_ERR"
                elif res.status in [429, 500, 503]:
                    return "BUSY"
                else:
                    return f"ERR_{res.status}"

    async def get_embedding(self, text):
        """Fetches vector embedding for text."""
        attempts = 0
        while attempts < 3:
            key = self.keyring.get_next()
            if not key: return None
            
            url = f"https://generativelanguage.googleapis.com/v1beta/models/{self.embed_model}:embedContent"
            payload = {"content": {"parts": [{"text": text}]}}
            
            res = await self._api_call(url, payload, key)
            
            if isinstance(res, dict) and 'embedding' in res:
                return res['embedding']['values']
            elif res == "AUTH_ERR":
                self.keyring.bury(key)
            elif res == "BUSY":
                await asyncio.sleep(1)
            
            attempts += 1
        return None

    async def invoke(self, prompt, sys_inst=None, temp=None):
        if temp is None: temp = self.settings.get("temperature_base", 0.8)
        
        for model in self.models:
            attempts = 0
            while attempts < 3:
                key = self.keyring.get_next()
                if not key: return f"{ICONS['error']} NO KEYS"

                gen_config = {"temperature": temp}
                if "gemini-3" in model: gen_config["thinking_level"] = "medium"

                url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent"
                payload = {
                    "contents": [{"parts": [{"text": prompt}]}],
                    "generationConfig": gen_config
                }
                if sys_inst: payload["system_instruction"] = {"parts": [{"text": sys_inst}]}

                res = await self._api_call(url, payload, key)

                if isinstance(res, dict) and 'candidates' in res:
                    return res['candidates'][0]['content']['parts'][0]['text'].strip()
                elif res == "AUTH_ERR":
                    self.keyring.bury(key)
                elif res == "BUSY":
                    await asyncio.sleep(1)
                
                attempts += 1
                
        return f"{ICONS['error']} SILENCE"

    async def _get_semantic_context(self, target_vec, text_fallback):
        """Finds semantically similar nodes in the graph."""
        state = await self.memory.load_graph()
        nodes = state.get('nodes', [])
        
        if not target_vec:
            # Fallback to keyword search if embedding failed
            return await self.memory.get_graph_context(text_fallback)
        
        # Calculate Cosine Similarity for all nodes with embeddings
        scored = []
        for n in nodes:
            if 'vector' in n:
                score = VectorMath.cosine_similarity(target_vec, n['vector'])
                if score > 0.5: # Min relevance
                    scored.append((score, n['label']))
        
        # Top 5 Semantic Matches
        scored.sort(key=lambda x: x[0], reverse=True)
        top_matches = scored[:5]
        
        if not top_matches: return "No semantic associations."
        return "Associations: " + "; ".join([f"{t[1]} ({t[0]:.2f})" for t in top_matches])

    async def _autonomous_think(self):
        """Generates ideas and links them ASSOCIATIVELY, not just chronologically."""
        if random.random() < 0.75 and self.settings.get("autonomy_enabled"):
            recent = await self.memory.recall_adaptive(limit=5)
            context = "\n".join([item[1] for item in recent])
            idea_prompt = f"Based on: {context}\nGenerate 1 abstract concept JSON: {{label, emoji}}"
            
            res = await self.face.invoke(idea_prompt)
            if "{" in res:
                try:
                    clean = res[res.find('{'):res.rfind('}')+1]
                    data = json.loads(clean)
                    
                    # 1. Vectorize the new idea
                    vector = await self.face.get_embedding(data['label'])
                    
                    graph = await self.memory.load_graph()
                    new_id = str(time.time())
                    
                    # 2. Find Parents (Associative Linking)
                    parents = []
                    if vector and graph['nodes']:
                        threshold = self.settings.get("semantic_threshold", 0.72)
                        for n in graph['nodes']:
                            if 'vector' in n:
                                sim = VectorMath.cosine_similarity(vector, n['vector'])
                                if sim > threshold:
                                    parents.append(n['id'])
                    
                    # Fallback parent (chronological) if no semantic link found
                    if not parents and graph['nodes']:
                        parents.append(graph['nodes'][-1]['id'])
                    
                    # 3. Add Node
                    new_node = {
                        "id": new_id,
                        "label": data.get('label', 'Idea'),
                        "type": "idea",
                        "emoji": data.get('emoji', "\U0001F4A1"),
                        "vector": vector # STORE EMBEDDING
                    }
                    graph.setdefault("nodes", []).append(new_node)
                    
                    # 4. Create Links (Multiple Parents!)
                    for pid in parents[:3]: # Limit to 3 strongest connections
                        graph.setdefault("links", []).append({"source": pid, "target": new_id})
                    
                    await self.memory.save_graph(graph["nodes"], graph["links"])
                    logger.info(f"{ICONS['link']} New Node Linked to {len(parents)} parents.")
                    
                except Exception as e: logger.error(f"Autonomy err: {e}")
