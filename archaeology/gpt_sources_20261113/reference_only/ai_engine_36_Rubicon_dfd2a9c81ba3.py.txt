class VectorMath:
    @staticmethod
    def cosine_similarity(v1, v2):
        if not v1 or not v2 or len(v1) != len(v2): return 0.0
        dot = sum(a * b for a, b in zip(v1, v2))
        mag1 = math.sqrt(sum(a * a for a in v1))
        mag2 = math.sqrt(sum(b * b for b in v2))
        if mag1 == 0 or mag2 == 0: return 0.0
        return dot / (mag1 * mag2)

class Keyring:
    def __init__(self, keys):
        self.keys = list(set(keys))
        env_key = os.environ.get("JANUS_API_KEY")
        if env_key: self.keys.insert(0, env_key)
        self.index = 0
        self.total = len(self.keys)
    def reload(self, new_keys):
        self.keys = list(set(self.keys) | set(new_keys))
        self.total = len(self.keys)
        self.index = 0
    def get_next(self):
        if self.total == 0: return None
        k = self.keys[self.index]
        self.index = (self.index + 1) % self.total
        return k
    def bury(self, key):
        if key in self.keys:
            self.keys.remove(key)
            self.total = len(self.keys)
            if self.total > 0: self.index = self.index % self.total
            logger.warning(f"{ICONS['skull']} Key buried. Left: {self.total}")

class FastMemory:
    def __init__(self, capacity=30, ttl=600):
        self.cache = cachetools.TTLCache(maxsize=capacity, ttl=ttl)
    def add(self, key, value): self.cache[key] = value
    def get_recent(self, limit=10):
        items = sorted(self.cache.items(), key=lambda x: x[0], reverse=True)
        return items[:limit]
    def clear(self): self.cache.clear()

            sql = f"SELECT source, content FROM thoughts WHERE {conditions} ORDER BY id DESC LIMIT ?"

            params.append(limit)

                await db.commit()
        except: pass

class JanusFace:
    def __init__(self, keyring, settings):
        self.keyring = keyring
        self.settings = settings
        self.models = [settings.get("model_primary"), settings.get("model_backup")]
        self.embed_model = settings.get("model_embedding", "text-embedding-004")

    async def _api_call(self, url, payload, key):
        try:
            async with aiohttp.ClientSession() as session:
                async with session.post(f"{url}?key={key}", json=payload, timeout=20) as res:
                    if res.status == 200: return await res.json()
                    elif res.status in [401, 403]: return "AUTH_ERR"
                    elif res.status in [429, 500, 503]: return "BUSY"
                    else: return f"ERR_{res.status}"
        except: return "NET_ERR"

    async def get_embedding(self, text):
        attempts = 0
        while attempts < 2:
            key = self.keyring.get_next()
            if not key: return None
            url = f"https://generativelanguage.googleapis.com/v1beta/models/{self.embed_model}:embedContent"
            payload = {"content": {"parts": [{"text": text}]}}
            res = await self._api_call(url, payload, key)
            if isinstance(res, dict) and 'embedding' in res: return res['embedding']['values']
            elif res == "AUTH_ERR": self.keyring.bury(key)
            elif res == "BUSY": await asyncio.sleep(1)
            attempts += 1
        return None

    async def invoke(self, prompt, sys_inst=None, temp=None):
        if temp is None: temp = self.settings.get("temperature_base", 0.8)
        for model in self.models:
            attempts = 0
            while attempts < 2:
                key = self.keyring.get_next()
                if not key: return f"{ICONS['error']} NO KEYS"
                gen_config = {"temperature": temp}
                if "gemini-3" in model: gen_config["thinking_level"] = "medium"
                url = f"https://generativelanguage.googleapis.com/v1beta/models/{model}:generateContent"
                payload = {"contents": [{"parts": [{"text": prompt}]}], "generationConfig": gen_config}
                if sys_inst: payload["system_instruction"] = {"parts": [{"text": sys_inst}]}
                res = await self._api_call(url, payload, key)
                if isinstance(res, dict) and 'candidates' in res: return res['candidates'][0]['content']['parts'][0]['text'].strip()
                elif res == "AUTH_ERR": self.keyring.bury(key)
                elif res == "BUSY": await asyncio.sleep(1)
                attempts += 1
        return f"{ICONS['error']} SILENCE"

class JanusNerveUDP(asyncio.DatagramProtocol):
    def __init__(self, nerve): self.nerve = nerve
    def datagram_received(self, data, addr):
        asyncio.create_task(self.nerve._ingest(data.decode('utf-8', 'ignore').strip(), "UDP", addr[0]))

class JanusNerve:
    def __init__(self, engine):
        self.engine = engine
        self.settings = engine.settings.data.get("nerve_config", {})
    async def start(self):
        try:
            port = self.settings.get('syslog_udp_port', 514)
            loop = asyncio.get_running_loop()
            await loop.create_datagram_endpoint(lambda: JanusNerveUDP(self), local_addr=('0.0.0.0', port))
            logger.info(f"{ICONS['nerve']} UDP Listener on {port}")
        except: pass
    async def _ingest(self, content, source_type, metadata=None):
        if not content: return
        try:
            for w in self.settings.get("filters", []): 
                if w in content.lower(): return
            await self.engine.memory.remember(f"ROUTER_{source_type}", content)
        except: pass
    async def _handle_http_log(self, request):
        try:
            data = await request.text()
            await self._ingest(data, "HTTP", request.remote)
            return web.Response(text="OK")
        except: return web.Response(status=500)

    async def _get_semantic_context(self, target_vec, text_fallback):
        state = await self.memory.load_graph()
        nodes = state.get('nodes', [])
        if not target_vec: return await self.memory.get_graph_context(text_fallback)
        scored = []
        for n in nodes:
            if 'vector' in n:
                score = VectorMath.cosine_similarity(target_vec, n['vector'])
                if score > 0.5: scored.append((score, n['label']))
        scored.sort(key=lambda x: x[0], reverse=True)
        top_matches = scored[:5]
        if not top_matches: return "Нет ассоциаций."
        return "Ассоциации: " + "; ".join([f"{t[1]} ({t[0]:.2f})" for t in top_matches])

                        except: continue

                        await db.execute("DELETE FROM vault WHERE key_type=?", (offset_key,))
                        await db.execute("INSERT INTO vault (key_type, value) VALUES (?, ?)", (offset_key, str(new_offset)))
